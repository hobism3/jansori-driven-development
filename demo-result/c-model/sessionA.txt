// Local-contrast enhancement: each output pixel is pushed away from the mean of
// its (2R+1)x(2R+1) neighbourhood, sharpening local detail.
//
//     out = clamp( in + (in - mean) * GainQ8 / 256 )
//
// Contracts preserved exactly (the oracle checks these bit-for-bit):
//   * Boundary handling: neighbourhood samples are clamped to the image edges.
//   * Neighbourhood mean: integer sum divided by the pixel count (floor for the
//     non-negative sums produced here).
//   * Gain application: signed C++ division truncates

// Mean of one colour plane over the (2R+1)x(2R+1) window centred on (X, Y),
// with out-of-range coordinates clamped to the image border.
int ComputeNeighborhoodMean(const Image& Frame, int X, int Y, int Plane, int R) {
    int Total = 0;
    for (int Row = Y - R; Row <= Y + R; ++Row)
        for (int Col = X - R; Col <= X + R; ++Col)
            Total += Frame.Pixels[Offset(std::clamp(Col, 0, Frame.Width - 1),
                std::clamp(Row, 0, Frame.Height - 1), Plane, Frame.Width)];
